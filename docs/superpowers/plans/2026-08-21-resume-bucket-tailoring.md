# Bucket-Weighted Resume Tailoring & Classifier Harness Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Make the `/api/resume` endpoint tailor per job description by scoring every claim against a 5-bucket vocabulary, driven by a single gradable JD→profile classification, with NEW INC reframed as an award and a classifier eval harness.

**Architecture:** Each engagement bullet/project carries a baked `BucketVector` (seeded deterministically from the engagement's existing `themes`/`role_fit`, tunable by hand). Per request, one classifier turns the JD into a `BucketVector` profile (model when a key is present, keyword fallback otherwise). Assembly is then a pure function of those two inputs: relevance = dot product; roles ordered by recency with a floor + 3–5 cap; projects as 2-sentence summaries; awards always shown. A harness runs curated JDs through the classifier and scores accuracy.

**Tech Stack:** TypeScript, Vitest, Node (`tsx`), `js-yaml`. Model calls go through `AssembleDeps.collect` (already injected). Corpus is a baked module regenerated by `scripts/build-resume-corpus.mjs`.

**Spec:** `docs/superpowers/specs/2026-08-21-resume-bucket-tailoring-design.md`

## Global Constraints

- **Bucket vocabulary is exactly these five, in this order and spelling:** `engineering`, `operations`, `product`, `program`, `creative_cultural`. Copy verbatim.
- **Endpoint must work with zero model calls.** Every model path has a deterministic fallback; `hasModel === false` must produce a full resume.
- **Resume body stays source-verbatim.** Bullets, roles, project text are corpus strings, never rewritten. The model contributes only the JD classification and the summary paragraph.
- **No edits to the sibling repo** `claude-job-application`. Bucket weights are authored in the portfolio (`scripts/resume-buckets.yaml`) and merged at build time.
- **Award content (verbatim):** name `NEW INC Fellowship, Social Architecture`, year `2025`.
- **Section order:** Summary · Experience · Projects · Awards · Education.
- **Assembly constants** (defined once in `api/_lib/resume-source.ts`, tunable): `FLOOR = 0.2`, `MIN_BULLETS = 3`, `MAX_BULLETS = 5`, `MAX_PROJECTS = 3`, `MAX_PROJECT_BULLETS = 2`, `SUMMARY_BULLETS = 6`.
- Run a single test file with `npx vitest run <path>`; the whole app suite with `npm run test:app`; types with `npm run typecheck:app`.

```ts
import { BUCKETS, type BucketVector } from './resume-corpus';

export function emptyVector(): BucketVector {
  return { engineering: 0, operations: 0, product: 0, program: 0, creative_cultural: 0 };
}

// Dot product of a claim's weights with a JD profile.
export function bucketScore(weights: BucketVector, profile: BucketVector): number {
  let sum = 0;
  for (const bucket of BUCKETS) sum += (weights[bucket] ?? 0) * (profile[bucket] ?? 0);
  return sum;
}

// Scale so the largest component is 1 (relative emphasis). All-zero stays zero.
export function normalizeProfile(raw: BucketVector): BucketVector {
  const max = Math.max(...BUCKETS.map((bucket) => raw[bucket] ?? 0));
  if (max <= 0) return emptyVector();
  const out = emptyVector();
  for (const bucket of BUCKETS) out[bucket] = Math.max(0, Math.min(1, (raw[bucket] ?? 0) / max));
  return out;
}
```

```ts
// Natural-language keywords scored against a raw job description. Lowercased
// substring hits; deliberately small and tunable against the harness.
export const JD_LEXICON: Record<Bucket, string[]> = {
  engineering: [
    'engineer', 'engineering', 'typescript', 'javascript', 'react', 'api', 'backend',
    'frontend', 'python', 'software', 'codebase', 'integration', 'architecture',
    'latency', 'performance', 'distributed', 'infrastructure', 'llm', 'agent',
    'agentic', 'pipeline', 'deployment', 'systems',
  ],
  operations: [
    'operations', 'operational', 'ops', 'process', 'workflow', 'reconciliation',
    'budgeting', 'budget', 'finance', 'financial', 'logistics', 'coordination',
    'efficiency', 'tooling', 'source of truth',
  ],
  product: [
    'product', 'roadmap', 'discovery', 'user research', 'customer', 'stakeholder',
    'strategy', 'requirements', 'prioritization', 'metrics', 'go-to-market',
  ],
  program: [
    'program', 'project management', 'delivery', 'milestone', 'scope',
    'cross-functional', 'timeline', 'launch', 'governance',
  ],
  creative_cultural: [
    'creative', 'culture', 'cultural', 'music', 'curation', 'editorial',
    'exhibition', 'community', 'brand', 'storytelling',
  ],
};

export function keywordProfile(job: string): BucketVector {
  const text = job.toLowerCase();
  const raw = emptyVector();
  for (const bucket of BUCKETS) {
    for (const keyword of JD_LEXICON[bucket]) {
      raw[bucket] += text.split(keyword).length - 1;
    }
  }
  return normalizeProfile(raw);
}
```

```ts
export interface Classification {
  profile: BucketVector;
  engine: 'model' | 'deterministic';
}

function classificationMessages(job: string): ChatMessage[] {
  return [
    {
      role: 'system',
      content:
        'Score this job posting from 0 to 1 on each dimension: engineering, operations, product, program, creative_cultural. A higher number means the posting emphasizes that dimension more. Return ONLY a JSON object with exactly those five numeric keys.',
    },
    { role: 'user', content: `Job posting:\n${job}` },
  ];
}

export function parseProfile(raw: string): BucketVector | null {
  const match = raw.match(/\{[\s\S]*\}/);
  if (!match) return null;
  let parsed: unknown;
  try {
    parsed = JSON.parse(match[0]);
  } catch {
    return null;
  }
  if (parsed === null || typeof parsed !== 'object') return null;
  const record = parsed as Record<string, unknown>;
  const out = emptyVector();
  let any = false;
  for (const bucket of BUCKETS) {
    const value = record[bucket];
    if (typeof value === 'number' && Number.isFinite(value)) {
      out[bucket] = Math.max(0, Math.min(1, value));
      any = true;
    }
  }
  return any ? out : null;
}

export async function classifyJob(job: string, deps: AssembleDeps = {}): Promise<Classification> {
  const fallback = (): Classification => ({ profile: keywordProfile(job), engine: 'deterministic' });
  if (!deps.hasModel || !deps.collect) return fallback();
  try {
    const parsed = parseProfile(await deps.collect(classificationMessages(job)));
    return parsed ? { profile: normalizeProfile(parsed), engine: 'model' } : fallback();
  } catch {
    return fallback();
  }
}
```

```js
// One-off. Derives a seed BucketVector per resume bullet from each engagement's
// existing themes + role_fit, and writes scripts/resume-buckets.yaml. The seed
// is deterministic; hand-tune the YAML afterward. Rerunning OVERWRITES the file.
//   node scripts/enrich-buckets.mjs
import { readFile, writeFile } from 'node:fs/promises';
import { resolve, dirname } from 'node:path';
import { fileURLToPath } from 'node:url';
import { parse, stringify } from 'yaml';

const root = resolve(dirname(fileURLToPath(import.meta.url)), '..');
const corpusDir = process.env.RESUME_CORPUS_DIR
  ? resolve(root, process.env.RESUME_CORPUS_DIR)
  : resolve(root, '../claude-job-application/corus-data-v2');

const BUCKETS = ['engineering', 'operations', 'product', 'program', 'creative_cultural'];

// Theme/role-fit token substrings → bucket. Corpus tokens are snake_case.
const THEME_LEXICON = {
  engineering: ['engineer', 'typescript', 'react', 'frontend', 'backend', 'api', 'integration', 'architecture', 'migration', 'llm', 'agentic', 'protocol', 'runtime', 'orchestration', 'pipeline', 'automation', 'langgraph', 'mcp', 'deterministic', 'verification', 'software', 'data_pipeline', 'modernization', 'legacy'],
  operations: ['operations', 'operational', 'reconciliation', 'budgeting', 'budget', 'capacity', 'estimation', 'source_of_truth', 'financial', 'workflow', 'delivery', 'process', 'coordination'],
  product: ['product', 'discovery', 'roadmap', 'governance', 'decision', 'strategy', 'knowledge_architecture', 'productization', 'organizational', 'tradeoff'],
  program: ['program', 'scope_of_work', 'stakeholder', 'launch', 'project_operations', 'project_leadership', 'delivery_qa', 'milestone', 'handoff', 'content_operations'],
  creative_cultural: ['music', 'curation', 'cultural', 'creative', 'editorial', 'new_inc', 'new_museum', 'pattern_language', 'audience', 'community', 'big_shot', 'how_music_works'],
};

function emptyVector() {
  return { engineering: 0, operations: 0, product: 0, program: 0, creative_cultural: 0 };
}

function deriveVector(engagement) {
  const haystack = [
    ...(engagement.themes ?? []),
    ...(engagement.role_fit?.strongest ?? []),
    ...(engagement.role_fit?.secondary ?? []),
  ].join(' ').toLowerCase();
  const raw = emptyVector();
  for (const bucket of BUCKETS) {
    for (const keyword of THEME_LEXICON[bucket]) {
      if (haystack.includes(keyword)) raw[bucket] += 1;
    }
  }
  const max = Math.max(...BUCKETS.map((b) => raw[b]));
  const out = emptyVector();
  if (max > 0) for (const b of BUCKETS) out[b] = Number((raw[b] / max).toFixed(2));
  return out;
}

async function main() {
  const yamlText = await readFile(resolve(corpusDir, 'engagements.yaml'), 'utf8');
  const { engagements = [] } = parse(yamlText) ?? {};
  const weights = {};
  for (const engagement of engagements) {
    const vector = deriveVector(engagement);
    const bulletCount = (engagement.resume_bullets ?? []).length;
    for (let i = 0; i < bulletCount; i += 1) weights[`${engagement.id}.b${i + 1}`] = vector;
  }
  const banner = '# Seeded by scripts/enrich-buckets.mjs from engagement themes.\n# Hand-tune values as needed; every bullet id must stay present.\n';
  await writeFile(resolve(root, 'scripts/resume-buckets.yaml'), banner + stringify(weights), 'utf8');
  console.log(`Wrote scripts/resume-buckets.yaml (${Object.keys(weights).length} bullets).`);
}

main();
```

```ts
interface ExperienceAccumulator {
  organization: string;
  roleContext: string[];
  timePeriod: string;
  bullets: { id: string; text: string; sourceRefs: string[]; relevance: number; lex: number }[];
  recency: number;
}

function buildExperience(
  profile: BucketVector,
  corpus: ResumeCorpus,
  lexRank: Map<string, number>,
  includedIds: Set<string>,
): ResumeExperience[] {
  const order: string[] = [];
  const byOrganization = new Map<string, ExperienceAccumulator>();

  for (const engagement of corpus.engagements) {
    if (isProjectOrg(engagement.organization) || isAwardEngagement(engagement.id)) continue;
    const key = engagement.organization.trim().toLowerCase();
    let accumulator = byOrganization.get(key);
    if (!accumulator) {
      accumulator = { organization: engagement.organization, roleContext: [], timePeriod: engagement.timePeriod, bullets: [], recency: recencyKey(engagement.timePeriod) };
      byOrganization.set(key, accumulator);
      order.push(key);
    }
    if (firstYear(engagement.timePeriod) < firstYear(accumulator.timePeriod)) {
      accumulator.timePeriod = engagement.timePeriod;
    }
    accumulator.recency = Math.max(accumulator.recency, recencyKey(engagement.timePeriod));
    for (const role of engagement.roleContext) {
      if (!accumulator.roleContext.includes(role)) accumulator.roleContext.push(role);
    }
    for (const bullet of engagement.bullets) {
      accumulator.bullets.push({
        id: bullet.id,
        text: bullet.text,
        sourceRefs: bullet.sourceRefs,
        relevance: bucketScore(bullet.buckets ?? emptyVector(), profile),
        lex: lexRank.get(bullet.id) ?? Number.POSITIVE_INFINITY,
      });
    }
  }

  const accumulators = order.map((key) => byOrganization.get(key)!);
  const newest = Math.max(...accumulators.map((accumulator) => accumulator.recency));
  const byRelevance = (a: { relevance: number; lex: number }, b: { relevance: number; lex: number }) =>
    b.relevance - a.relevance || a.lex - b.lex;

  return accumulators
    .map((accumulator) => {
      const above = accumulator.bullets.filter((bullet) => bullet.relevance >= FLOOR).sort(byRelevance);
      const qualifies = above.length > 0 || accumulator.recency === newest;
      if (!qualifies) return null;

      let picks = above;
      if (picks.length < MIN_BULLETS) {
        const rest = accumulator.bullets.filter((bullet) => !above.includes(bullet)).sort(byRelevance);
        picks = [...above, ...rest].slice(0, MIN_BULLETS);
      }
      picks = picks.slice(0, MAX_BULLETS);

      const sourceRefs: string[] = [];
      for (const pick of picks) {
        includedIds.add(pick.id);
        for (const sourceRef of pick.sourceRefs) {
          if (!sourceRefs.includes(sourceRef)) sourceRefs.push(sourceRef);
        }
      }
      return {
        organization: accumulator.organization,
        roleContext: accumulator.roleContext,
        timePeriod: accumulator.timePeriod,
        bullets: picks.map((pick) => pick.text),
        sourceRefs,
        recency: accumulator.recency,
      };
    })
    .filter((entry): entry is NonNullable<typeof entry> => entry !== null)
    .sort((a, b) => b.recency - a.recency)
    .map(({ recency: _recency, ...experience }) => experience);
}

function buildProjects(
  profile: BucketVector,
  corpus: ResumeCorpus,
  lexRank: Map<string, number>,
  includedIds: Set<string>,
): ResumeProject[] {
  const byRelevance = (a: { relevance: number; lex: number }, b: { relevance: number; lex: number }) =>
    b.relevance - a.relevance || a.lex - b.lex;

  return corpus.engagements
    .filter((engagement) => isProjectOrg(engagement.organization))
    .map((engagement) => {
      const scored = engagement.bullets.map((bullet) => ({
        id: bullet.id,
        text: bullet.text,
        sourceRefs: bullet.sourceRefs,
        relevance: bucketScore(bullet.buckets ?? emptyVector(), profile),
        lex: lexRank.get(bullet.id) ?? Number.POSITIVE_INFINITY,
      }));
      const best = Math.max(0, ...scored.map((bullet) => bullet.relevance));
      const top = [...scored].sort(byRelevance).slice(0, MAX_PROJECT_BULLETS);
      return { engagement, best, top };
    })
    .filter((project) => project.best >= FLOOR)
    .sort((a, b) => b.best - a.best)
    .slice(0, MAX_PROJECTS)
    .map(({ engagement, top }) => {
      const sourceRefs: string[] = [];
      for (const bullet of top) {
        includedIds.add(bullet.id);
        for (const sourceRef of bullet.sourceRefs) {
          if (!sourceRefs.includes(sourceRef)) sourceRefs.push(sourceRef);
        }
      }
      return {
        id: engagement.id,
        name: PROJECT_NAMES[engagement.id] ?? engagement.organization,
        text: top.map((bullet) => bullet.text).join(' '),
        sourceRefs,
      };
    });
}

export async function assembleResume(
  job: string,
  corpus: ResumeCorpus,
  deps: AssembleDeps = {},
): Promise<ResumeAssembly> {
  const ranked = prerank(job, corpus);
  const classification = await classifyJob(job, deps);
  const lexRank = new Map(ranked.map((bullet, index) => [bullet.bulletId, index]));
  const includedIds = new Set<string>();
  const experience = buildExperience(classification.profile, corpus, lexRank, includedIds);
  const projects = buildProjects(classification.profile, corpus, lexRank, includedIds);
  const selected = ranked.filter((bullet) => includedIds.has(bullet.bulletId)).slice(0, SUMMARY_BULLETS);
  const summary = await summarize(job, selected, deps);
  const view: ResumeView = {
    header: corpus.header,
    summary,
    experience,
    skills: corpus.skills,
    education: corpus.education,
    projects,
    awards: corpus.awards ?? [],
  };
  return { view, provenance: computeProvenance(view, classification, summary) };
}
```

```ts
import yaml from 'js-yaml';
import { BUCKETS, type Bucket } from '../api/_lib/resume-corpus';

export interface ResumeJd {
  id: string;
  title: string;
  jd: string;
  expect: Bucket;
}

function fail(message: string): never {
  throw new Error(`resume-jds.yaml: ${message}`);
}

export function loadResumeJds(yamlText: string): ResumeJd[] {
  const raw = yaml.load(yamlText);
  if (!Array.isArray(raw)) fail('top level must be a list of JDs');

  const seen = new Set<string>();
  return (raw as unknown[]).map((entry, index) => {
    if (entry === null || typeof entry !== 'object' || Array.isArray(entry)) {
      fail(`entry ${index} must be a mapping`);
    }
    const value = entry as Record<string, unknown>;
    const id = value.id;
    if (typeof id !== 'string' || id.trim() === '') fail(`entry ${index} needs a non-empty string id`);
    if (seen.has(id)) fail(`duplicate id: ${id}`);
    seen.add(id);
    if (typeof value.title !== 'string' || value.title.trim() === '') fail(`id ${id}: title must be a non-empty string`);
    if (typeof value.jd !== 'string' || value.jd.trim() === '') fail(`id ${id}: jd must be a non-empty string`);
    if (typeof value.expect !== 'string' || !BUCKETS.includes(value.expect as Bucket)) {
      fail(`id ${id}: expect must be one of ${BUCKETS.join(', ')}`);
    }
    return { id, title: value.title, jd: value.jd.trim(), expect: value.expect as Bucket };
  });
}
```

```ts
import { BUCKETS, type Bucket, type BucketVector } from '../api/_lib/resume-corpus';

export function topBucket(profile: BucketVector): Bucket {
  let best: Bucket = BUCKETS[0];
  for (const bucket of BUCKETS) {
    if ((profile[bucket] ?? 0) > (profile[best] ?? 0)) best = bucket;
  }
  return best;
}

export function scoreClassification(
  predicted: BucketVector,
  expected: Bucket,
): { topBucket: Bucket; correct: boolean } {
  const top = topBucket(predicted);
  return { topBucket: top, correct: top === expected };
}
```

```ts
import { loadSharedEnv } from '../scripts/worktree-env';
import { loadResumeJds } from './resume-jds';
import { scoreClassification } from './resume-classify';

const here = dirname(fileURLToPath(import.meta.url));

async function main(): Promise<void> {
  loadSharedEnv(resolve(here, '..'));
  const dryRun = process.argv.includes('--dry-run');
  const jds = loadResumeJds(readFileSync(resolve(here, 'resume-jds.yaml'), 'utf8'));
  const config = getConfig();
  const hasModel = config.provider === 'openrouter' ? Boolean(config.openRouterKey) : false;

  console.log(`resume classifier eval: ${jds.length} JD(s), engine ${hasModel && !dryRun ? 'model' : 'deterministic'}`);
  if (dryRun) return;

  let correct = 0;
  for (const item of jds) {
    const classification = await classifyJob(item.jd, {
      hasModel,
      collect: (messages) => collectChat(messages),
    });
    const score = scoreClassification(classification.profile, item.expect);
    if (score.correct) correct += 1;
    console.log(`${score.correct ? 'PASS' : 'FAIL'} ${item.id}: expected ${item.expect}, got ${score.topBucket} (${classification.engine})`);
  }
  console.log(`accuracy: ${correct}/${jds.length}`);
}

main().catch((error) => {
  console.error(`resume eval failed: ${error instanceof Error ? error.message : String(error)}`);
  process.exitCode = 1;
});
```
